! lệnh tạo bảng.
PRAGMA foreign_keys = ON;

-- =============================================
-- 1. BẢNG PHÒNG KINH DOANH
-- =============================================
CREATE TABLE PhongKinhDoanh (
    MaPhong TEXT PRIMARY KEY,
    TenPhong TEXT NOT NULL UNIQUE
);

-- =============================================
-- 2. BẢNG MÔI GIỚI
-- =============================================
CREATE TABLE MoiGioi (
    MaMoiGioi TEXT PRIMARY KEY,
    HoTen TEXT NOT NULL,
    SoDienThoai TEXT UNIQUE,
    Email TEXT UNIQUE,
    NgayVaoLam DATE NOT NULL,
    MaPhong TEXT,
    FOREIGN KEY (MaPhong) REFERENCES PhongKinhDoanh(MaPhong)
);

-- =============================================
-- 3. BẢNG CHỦ SỞ HỮU
-- =============================================
CREATE TABLE ChuSoHuu (
    MaChu TEXT PRIMARY KEY,
    HoTen TEXT NOT NULL,
    CCCD TEXT UNIQUE,
    SoDienThoai TEXT UNIQUE,
    Email TEXT UNIQUE
);

-- =============================================
-- 4. BẢNG KHÁCH HÀNG
-- =============================================
CREATE TABLE KhachHang (
    MaKhachHang TEXT PRIMARY KEY,
    HoTen TEXT NOT NULL,
    SoDienThoai TEXT UNIQUE,
    Email TEXT UNIQUE
);

-- =============================================
-- 5. BẢNG BẤT ĐỘNG SẢN
-- =============================================
CREATE TABLE BatDongSan (
    MaBDS TEXT PRIMARY KEY,
    MaChu TEXT NOT NULL,
    LoaiBDS TEXT NOT NULL,
    DiaChi TEXT NOT NULL,
    QuanHuyen TEXT NOT NULL,
    DienTich REAL NOT NULL CHECK (DienTich > 0),
    GiaBan REAL NOT NULL CHECK (GiaBan > 0),
    SoGoiGCN TEXT NOT NULL UNIQUE,
    SoPhongNgu INTEGER DEFAULT 0 CHECK (SoPhongNgu >= 0),
    TrangThai TEXT NOT NULL,
    FOREIGN KEY (MaChu) REFERENCES ChuSoHuu(MaChu)
);

-- =============================================
-- 6. BẢNG HỢP ĐỒNG KÝ GỬI
-- =============================================
CREATE TABLE HopDongKyGui (
    MaHDKG TEXT PRIMARY KEY,
    MaBDS TEXT NOT NULL,
    MaMoiGioi TEXT NOT NULL,
    LoaiKyGui TEXT NOT NULL,
    NgayBatDau DATE NOT NULL,
    NgayKetThuc DATE NOT NULL,
    TrangThai TEXT NOT NULL,
    CHECK (NgayKetThuc >= NgayBatDau),
    FOREIGN KEY (MaBDS) REFERENCES BatDongSan(MaBDS),
    FOREIGN KEY (MaMoiGioi) REFERENCES MoiGioi(MaMoiGioi)
);

-- =============================================
-- 7. BẢNG LỊCH XEM NHÀ
-- =============================================
CREATE TABLE LichXemNha (
    MaLichXem TEXT PRIMARY KEY,
    MaKhachHang TEXT,
    MaBDS TEXT,
    MaMoiGioi TEXT,
    ThoiGianXem DATETIME NOT NULL,
    PhanHoi TEXT,
    FOREIGN KEY (MaKhachHang) REFERENCES KhachHang(MaKhachHang),
    FOREIGN KEY (MaBDS) REFERENCES BatDongSan(MaBDS),
    FOREIGN KEY (MaMoiGioi) REFERENCES MoiGioi(MaMoiGioi)
);

-- =============================================
-- 8. BẢNG HỢP ĐỒNG ĐẶT CỌC
-- =============================================
CREATE TABLE HopDongDatCoc (
    MaHDDC TEXT PRIMARY KEY,
    MaKhachHang TEXT,
    MaBDS TEXT,
    NgayDatCoc DATE NOT NULL,
    SoTienCoc REAL NOT NULL CHECK (SoTienCoc > 0),
    NgayHetHan DATE NOT NULL,
    TrangThai TEXT NOT NULL,
    CHECK (NgayHetHan >= NgayDatCoc),
    FOREIGN KEY (MaKhachHang) REFERENCES KhachHang(MaKhachHang),
    FOREIGN KEY (MaBDS) REFERENCES BatDongSan(MaBDS)
);

-- =============================================
-- 9. BẢNG GIAO DỊCH
-- =============================================
CREATE TABLE GiaoDich (
    MaGiaoDich TEXT PRIMARY KEY,
    MaBDS TEXT,
    MaKhachHang TEXT,
    MaMoiGioi TEXT,
    MaHDDC TEXT UNIQUE,
    LoaiGiaoDich TEXT CHECK (LoaiGiaoDich IN ('Mua ban', 'Cho thue')),
    NgayGiaoDich DATE,
    GiaTriGiaoDich REAL CHECK (GiaTriGiaoDich > 0),
    TrangThai TEXT,
    FOREIGN KEY (MaBDS) REFERENCES BatDongSan(MaBDS),
    FOREIGN KEY (MaKhachHang) REFERENCES KhachHang(MaKhachHang),
    FOREIGN KEY (MaMoiGioi) REFERENCES MoiGioi(MaMoiGioi),
    FOREIGN KEY (MaHDDC) REFERENCES HopDongDatCoc(MaHDDC)
);

-- =============================================
-- 10. BẢNG HOA HỒNG
-- =============================================
CREATE TABLE HoaHong (
    MaHoaHong TEXT PRIMARY KEY,
    MaGiaoDich TEXT,
    PhanTramHoaHong REAL CHECK (PhanTramHoaHong BETWEEN 0.5 AND 10),
    SoTienHoaHong REAL CHECK (SoTienHoaHong >= 0),
    NgayThu DATE,
    TrangThai TEXT,
    FOREIGN KEY (MaGiaoDich) REFERENCES GiaoDich(MaGiaoDich)
);

-- =============================================
-- 11. BẢNG HỢP ĐỒNG MUA BÁN
-- =============================================
CREATE TABLE HopDongMuaBan (
    MaHDMB TEXT PRIMARY KEY,
    MaGiaoDich TEXT NOT NULL UNIQUE,
    NgayKy DATE,
    GiaTriHopDong REAL CHECK (GiaTriHopDong > 0),
    PhuongThucThanhToan TEXT,
    FOREIGN KEY (MaGiaoDich) REFERENCES GiaoDich(MaGiaoDich)
);

-- =============================================
-- 12. BẢNG HỢP ĐỒNG THUÊ
-- =============================================
CREATE TABLE HopDongThue (
    MaHDT TEXT PRIMARY KEY,
    MaGiaoDich TEXT NOT NULL UNIQUE,
    NgayKy DATE,
    NgayBatDau DATE,
    NgayKetThuc DATE,
    TienThueThang REAL CHECK (TienThueThang > 0),
    CHECK (NgayKetThuc > NgayBatDau),
    FOREIGN KEY (MaGiaoDich) REFERENCES GiaoDich(MaGiaoDich)
);

-- =============================================
-- 13. BẢNG TIN ĐĂNG
-- =============================================
CREATE TABLE TinDang (
    MaTin TEXT PRIMARY KEY,
    MaBDS TEXT,
    MaMoiGioi TEXT,
    NgayDang DATE,
    FOREIGN KEY (MaBDS) REFERENCES BatDongSan(MaBDS),
    FOREIGN KEY (MaMoiGioi) REFERENCES MoiGioi(MaMoiGioi)
);
